# Install into a venv created WITH system site packages, so the apt-provided
# picamera2 / libcamera / python3-av / numpy / gpiozero / pyserial are used:
#   python3 -m venv --system-site-packages venv
#   venv/bin/pip install -c <(echo "numpy==$(python3 -c 'import numpy;print(numpy.__version__)')") -r requirements.txt
# The constraint stops pip from replacing the system numpy that picamera2 was
# built against. On samwise (Pi 5, Pi OS trixie, Python 3.13) the system
# numpy is 2.2.4.
# Also needed, not from pip: MediaMTX (github.com/bluenviron/mediamtx, v1.21.4
# linux_arm64) at ./mediamtx/mediamtx, $MEDIAMTX_BIN, or on PATH; ffmpeg.

# homography_calibration.py auto-detects and falls back to AKAZE otherwise.
opencv-python-headless>=4.8,<5
